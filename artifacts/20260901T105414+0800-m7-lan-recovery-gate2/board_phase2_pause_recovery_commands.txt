sh <<'M7_GATE2_PHASE2_PAUSE_RECOVERY'
set +e
umask 077
RUN_ROOT=/var/lib/gatewayd-m7-test-20260901T105414
EVIDENCE=$RUN_ROOT/evidence
CONFIG=$RUN_ROOT/main/gateway.conf
SPOOL=$RUN_ROOT/main/spool.data
STATE=$RUN_ROOT/main/spool.state
EXPECTED_SIZE=2744720
EXPECTED_SPOOL_HASH=0465c82da8da6bcbdd4b7cba44361197279fe3519c8d1a37a8a897eba241bbbf
EXPECTED_STATE_HASH=59d3246e7632a423ed3fbb9aa53d42fa0ddf2e7f63b8519927d7cc1e1f6a0d97
EXPECTED_RX=7047547
OLD_PID=$(cat "$EVIDENCE/phase1_gateway.pid" 2>/dev/null)
PID=$(cat "$EVIDENCE/phase2_gateway.pid" 2>/dev/null)

fail() { echo "M7_GATE2_PHASE2_PAUSE_RECOVERY_FAIL: $*" >&2; exit 1; }

test -e "$EVIDENCE/kill9_once.marker" || fail 'one-shot SIGKILL marker is absent'
test -s "$EVIDENCE/phase1_kill9_exit.txt" || fail 'SIGKILL exit evidence is absent'
grep -q '^kill9_exit=0$' "$EVIDENCE/phase1_kill9_exit.txt" || fail 'recorded SIGKILL did not succeed'
test -s "$EVIDENCE/phase1_wait_exit.txt" || fail 'phase1 wait evidence is absent'
test ! -e "/proc/$OLD_PID" || fail 'old phase1 PID still exists'
test -n "$PID" || fail 'missing phase2 gateway PID'
test -r "/proc/$PID/comm" || fail 'phase2 gateway PID is not alive'
grep -qx gatewayd "/proc/$PID/comm" || fail 'phase2 PID is not gatewayd'
tr '\000' ' ' < "/proc/$PID/cmdline" > "$EVIDENCE/phase2_recovery_process_check.txt"
echo >> "$EVIDENCE/phase2_recovery_process_check.txt"
grep -Fq "$CONFIG" "$EVIDENCE/phase2_recovery_process_check.txt" || fail 'phase2 PID does not reference this run config'

COUNT=0
while ! grep -q 'offline, durable records retained' "$EVIDENCE/phase2_gateway_stderr.log" 2>/dev/null && [ "$COUNT" -lt 15 ]; do
    test -r "/proc/$PID/comm" || fail 'phase2 exited while waiting for offline log'
    sleep 1
    COUNT=$((COUNT + 1))
done
grep -q 'offline, durable records retained' "$EVIDENCE/phase2_gateway_stderr.log" || fail 'phase2 offline log still absent after 15 seconds'

SIZE=$(wc -c < "$SPOOL")
SPOOL_HASH=$(sha256sum "$SPOOL" | awk '{print $1}')
STATE_HASH=$(sha256sum "$STATE" | awk '{print $1}')
RX=$(cat /sys/class/net/can0/statistics/rx_packets)
test "$SIZE" -eq "$EXPECTED_SIZE" || fail 'spool size changed while Broker and CAN source were offline'
test "$SPOOL_HASH" = "$EXPECTED_SPOOL_HASH" || fail 'spool hash changed after phase2 restart'
test "$STATE_HASH" = "$EXPECTED_STATE_HASH" || fail 'state hash changed after phase2 restart'
test "$RX" -eq "$EXPECTED_RX" || fail 'external CAN source is no longer paused'

CURRENT_STATE=$(awk '/^State:/ {print $2}' "/proc/$PID/status")
if [ "$CURRENT_STATE" != T ]; then
    echo "command=kill -STOP $PID" > "$EVIDENCE/phase2_pause_recovery_command.txt"
    kill -STOP "$PID"
    PAUSE_EXIT=$?
else
    echo "command=NOT RUN; phase2 was already stopped" > "$EVIDENCE/phase2_pause_recovery_command.txt"
    PAUSE_EXIT=0
fi
echo "sigstop_exit=$PAUSE_EXIT" > "$EVIDENCE/phase2_pause_recovery_exit.txt"
test "$PAUSE_EXIT" -eq 0 || fail 'phase2 SIGSTOP failed'
sleep 1
grep -Eq '^State:[[:space:]]+T' "/proc/$PID/status" || fail 'phase2 is not stopped'

{
    date -Ins
    echo "old_pid=$OLD_PID"
    echo "new_pid=$PID"
    cat "$EVIDENCE/phase1_kill9_command.txt"
    cat "$EVIDENCE/phase1_kill9_exit.txt"
    cat "$EVIDENCE/phase1_wait_exit.txt"
    cat "$EVIDENCE/phase2_pause_recovery_command.txt"
    cat "$EVIDENCE/phase2_pause_recovery_exit.txt"
    grep '^State:' "/proc/$PID/status"
    echo "spool_size=$SIZE"
    echo "spool_entries=$((SIZE / 80))"
    echo "rx_packets=$RX"
    ls -l "$SPOOL" "$STATE"
    sha256sum "$SPOOL" "$STATE"
    tail -n 30 "$EVIDENCE/phase2_gateway_stderr.log"
} > "$EVIDENCE/phase2_pause_recovery_result.txt" 2>&1
sync

cat "$EVIDENCE/phase2_pause_recovery_result.txt" 2>&1
echo "M7_GATE2_PHASE2_PAUSE_RECOVERY_PASS old_pid=$OLD_PID new_pid=$PID"
M7_GATE2_PHASE2_PAUSE_RECOVERY
