sh <<'M7_GATE2_POST_RECONNECT_DRAIN'
set +e
umask 077
RUN_ROOT=/var/lib/gatewayd-m7-test-20260901T105414
EVIDENCE=$RUN_ROOT/evidence
CONFIG=$RUN_ROOT/main/gateway.conf
SPOOL=$RUN_ROOT/main/spool.data
STATE=$RUN_ROOT/main/spool.state
MONITOR=$EVIDENCE/post_reconnect_input_drain_monitor.txt
BASE_SIZE=2744720
BASE_ENTRIES=34309
BASE_RX=7047547
PID=$(cat "$EVIDENCE/phase2_gateway.pid" 2>/dev/null)

fail() { echo "M7_GATE2_POST_RECONNECT_DRAIN_FAIL: $*" >&2; exit 1; }

read_state_u64()
{
    OFFSET=$1
    set -- $(dd if="$STATE" bs=1 skip="$OFFSET" count=8 2>/dev/null | od -An -v -tu1)
    test "$#" -eq 8 || return 1
    echo $(( $1 + $2 * 256 + $3 * 65536 + $4 * 16777216 + $5 * 4294967296 + $6 * 1099511627776 + $7 * 281474976710656 + $8 * 72057594037927936 ))
}

test -n "$PID" || fail 'missing phase2 gateway PID'
test -r "/proc/$PID/comm" || fail 'phase2 gateway PID is not alive'
grep -qx gatewayd "/proc/$PID/comm" || fail 'phase2 PID is not gatewayd'
tr '\000' ' ' < "/proc/$PID/cmdline" | grep -Fq "$CONFIG" || fail 'phase2 PID does not reference this run config'
test ! -e "$MONITOR" || fail 'refusing to overwrite post-reconnect monitor'

INITIAL_RX=$(cat /sys/class/net/can0/statistics/rx_packets)
test "$INITIAL_RX" -gt "$BASE_RX" || fail 'no post-reconnect physical CAN frames were observed'
DELTA_RX=$((INITIAL_RX - BASE_RX))
test "$DELTA_RX" -ge 300 || fail 'post-reconnect physical CAN window is too small'

PREV_SIZE=-1
PREV_RX=-1
STABLE=0
COUNT=0
{
    echo "monitor_started=$(date -Ins)"
    echo "gateway_pid=$PID"
    echo "base_spool_size=$BASE_SIZE"
    echo "base_entries=$BASE_ENTRIES"
    echo "base_rx_packets=$BASE_RX"
    echo "initial_rx_packets=$INITIAL_RX"
    echo "initial_rx_delta=$DELTA_RX"
} > "$MONITOR"

while [ "$COUNT" -lt 180 ]; do
    test -r "/proc/$PID/comm" || fail 'phase2 exited during post-reconnect drain'
    SIZE=$(wc -c < "$SPOOL")
    RX=$(cat /sys/class/net/can0/statistics/rx_packets)
    ACK_OFFSET=$(read_state_u64 16) || fail 'cannot decode state ack_offset'
    ACK_SEQ=$(read_state_u64 24) || fail 'cannot decode state ack_seq'
    NEXT_BATCH=$(read_state_u64 32) || fail 'cannot decode state next_batch'
    echo "sample=$COUNT time=$(date -Ins) spool_size=$SIZE rx_packets=$RX ack_offset=$ACK_OFFSET ack_seq=$ACK_SEQ next_batch=$NEXT_BATCH stable=$STABLE" >> "$MONITOR"
    if [ "$SIZE" = "$PREV_SIZE" ] && [ "$RX" = "$PREV_RX" ] && [ "$ACK_OFFSET" -eq "$SIZE" ]; then
        STABLE=$((STABLE + 1))
    else
        STABLE=0
    fi
    if [ "$STABLE" -ge 2 ]; then
        break
    fi
    PREV_SIZE=$SIZE
    PREV_RX=$RX
    sleep 2
    COUNT=$((COUNT + 1))
done

test "$STABLE" -ge 2 || fail 'CAN RX/spool/ACK did not become stable within 360 seconds'
test "$SIZE" -gt "$BASE_SIZE" || fail 'spool did not append post-reconnect records'
test $((SIZE % 80)) -eq 0 || fail 'spool size is not entry aligned'
ENTRIES=$((SIZE / 80))
NEW_ENTRIES=$((ENTRIES - BASE_ENTRIES))
test "$NEW_ENTRIES" -ge 300 || fail 'too few post-reconnect durable records'
test "$ACK_OFFSET" -eq "$SIZE" || fail 'ACK cursor did not reach spool end'
test "$ACK_SEQ" -eq "$ENTRIES" || fail 'gateway sequence is not contiguous after SIGKILL recovery'
grep -q 'network poll: offline, durable records retained' "$EVIDENCE/phase2_gateway_stderr.log" || fail 'runtime outage evidence is absent'
test "$(grep -c 'connected broker=' "$EVIDENCE/phase2_gateway_stderr.log")" -ge 2 || fail 'same-process reconnect evidence is absent'
sync

{
    date -Ins
    echo "gateway_pid=$PID"
    echo "base_rx_packets=$BASE_RX"
    echo "final_rx_packets=$RX"
    echo "rx_delta=$((RX - BASE_RX))"
    echo "base_spool_size=$BASE_SIZE"
    echo "final_spool_size=$SIZE"
    echo "base_entries=$BASE_ENTRIES"
    echo "final_entries=$ENTRIES"
    echo "new_entries=$NEW_ENTRIES"
    echo "ack_offset=$ACK_OFFSET"
    echo "ack_seq=$ACK_SEQ"
    echo "next_batch=$NEXT_BATCH"
    grep '^State:' "/proc/$PID/status"
    ls -l "$SPOOL" "$STATE"
    sha256sum "$SPOOL" "$STATE"
    ip -details -statistics link show can0
    tail -n 60 "$EVIDENCE/phase2_gateway_stderr.log"
} > "$EVIDENCE/post_reconnect_input_drained.txt" 2>&1

echo '--- post-reconnect drain monitor tail ---'
tail -n 30 "$MONITOR" 2>&1
echo '--- post-reconnect input drained ---'
cat "$EVIDENCE/post_reconnect_input_drained.txt" 2>&1
echo "M7_GATE2_POST_RECONNECT_DRAIN_PASS gateway_pid=$PID final_seq=$ACK_SEQ new_entries=$NEW_ENTRIES"
M7_GATE2_POST_RECONNECT_DRAIN
